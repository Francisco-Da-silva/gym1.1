CREATE TABLE Pagos (
    IdPago       INT IDENTITY(1,1) PRIMARY KEY,
    IdCliente    INT NOT NULL,
    FechaPago    DATE NOT NULL,
    FechaDesde   DATE NOT NULL,   -- desde cuándo cubre
    FechaHasta   DATE NOT NULL,   -- hasta cuándo cubre (vencimiento)
    Monto        DECIMAL(10,2) NOT NULL,
    Observacion  NVARCHAR(200) NULL,

    CONSTRAINT FK_Pagos_Clientes
        FOREIGN KEY (IdCliente) REFERENCES Clientes(IdCliente)
);

END
GO


CREATE PROCEDURE Estado_Pagos_Clientes
AS
BEGIN
    SET NOCOUNT ON;

    ;WITH UltimoPago AS (
        SELECT 
            c.IdCliente,
            c.Nombre,
            c.Apellido,
            c.PlanPago,
            MAX(p.FechaHasta) AS UltimaFechaHasta
        FROM Clientes c
        LEFT JOIN Pagos p ON c.IdCliente = p.IdCliente
        GROUP BY c.IdCliente, c.Nombre, c.Apellido, c.PlanPago
    )
    SELECT 
        IdCliente,
        Nombre,
        Apellido,
        PlanPago,
        UltimaFechaHasta,
        CASE 
            WHEN UltimaFechaHasta IS NULL THEN 'SIN PAGO'
            WHEN UltimaFechaHasta < CAST(GETDATE() AS DATE) THEN 'VENCIDO'
            WHEN DATEDIFF(DAY, CAST(GETDATE() AS DATE), UltimaFechaHasta) <= 5 THEN 'POR VENCER'
            ELSE 'AL DÍA'
        END AS EstadoPago,
        DATEDIFF(DAY, CAST(GETDATE() AS DATE), UltimaFechaHasta) AS DiasHastaVto
    FROM UltimoPago;
END
GO

CREATE OR ALTER PROCEDURE Listar_Pagos_Por_Cliente
    @IdCliente INT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT
        FechaPago,
        FechaDesde,
        FechaHasta,
        Monto,
        Observacion,
        -- Estado calculado según la fecha de vencimiento
        CASE 
            WHEN FechaHasta < CAST(GETDATE() AS DATE) THEN 'VENCIDO'
            WHEN DATEDIFF(DAY, CAST(GETDATE() AS DATE), FechaHasta) <= 5 THEN 'POR VENCER'
            ELSE 'AL DÍA'
        END AS EstadoPago
    FROM Pagos
    WHERE IdCliente = @IdCliente
    ORDER BY FechaHasta DESC;   -- 👉 el primero será el más reciente
END
GO
